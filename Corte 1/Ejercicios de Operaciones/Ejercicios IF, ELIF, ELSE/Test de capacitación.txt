#include <iostream>
using namespace std;
int main() 
{
    double totaldepreguntas, contestadascorrectamente;
    
    cout << "Introduce el total de las preguntas: "; 
    cin >> totaldepreguntas;
    cout << "Introduce las contestadas correctamente: ";
    cin >> contestadascorrectamente;
    
    double promedio = (contestadascorrectamente/totaldepreguntas)*100;
    
    cout << "Cantidad de respuestas correctamente respondidas: " << promedio;

        cout << "\nNivel maximo\n";

}
