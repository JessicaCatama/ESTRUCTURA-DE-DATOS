// 1 Example program
#include <iostream>
#include <string>
using namespace std;
int main()
{
    int numero1 = 8;
    int numero2 = 6;
    int producto = numero1 * numero2;
    std::cout << "El producto de " << numero1 << " y " << numero2 << " es: " << producto << std::endl;
    return 0;
}
